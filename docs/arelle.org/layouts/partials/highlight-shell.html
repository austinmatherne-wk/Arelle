{{- $code := .code -}} {{- $code = replaceRE "\r\n?" "\n" $code -}} {{- $lines := split $code "\n" -}} {{- $resultLines := slice -}} {{- range $lines -}} {{- $line := htmlEscape . -}} {{- if $line -}} {{- /* 1. Line continuation backslash at end of line */ -}} {{- $line = replaceRE `(\\)$` `$1` $line -}} {{- /* 2. Prompt at start of line */ -}} {{- $line = replaceRE `^(\$\s+)` `$1` $line -}} {{- /* 3. Quoted strings with variables */ -}} {{- $line = replaceRE `("[^&]*"|"[^&]*")` `$1` $line -}} {{- /* 4. Environment variables (like $PWD) */ -}} {{- $line = replaceRE `(\$[A-Z0-9_]+)` `$1` $line -}} {{- /* 5. Flags/options (run twice to handle consecutive flags like --rm -v) */ -}} {{- $line = replaceRE `(^|\s)(--[a-zA-Z0-9_-]+|-[a-zA-Z0-9]+)(\s|=|$|"|")` `$1$2$3` $line -}} {{- $line = replaceRE `(^|\s)(--[a-zA-Z0-9_-]+|-[a-zA-Z0-9]+)(\s|=|$|"|")` `$1$2$3` $line -}} {{- /* 6. Main executables/commands (standalone words only, not inside package/image names) */ -}} {{- $line = replaceRE `(^|\s|>)(arelleCmdLine(?:\.py)?|docker|pip3?|python3?)(?:\.exe)?(\s|$)` `$1$2$3` $line -}} {{- $line = replaceRE `(^|\s|>)(arelleCmdLine(?:\.py)?|docker|pip3?|python3?)(?:\.exe)?(\s|$)` `$1$2$3` $line -}} {{- /* 7. Subcommands */ -}} {{- $line = replaceRE `(^|\s)(run|install|pull|build|exec)(\s|$)` `$1$2$3` $line -}} {{- end -}} {{- $resultLines = $resultLines | append $line -}} {{- end -}} {{- delimit $resultLines "\n" | safeHTML -}}
